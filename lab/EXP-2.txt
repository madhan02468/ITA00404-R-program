# 5x4 Matrix (Filled by Rows)
mat1 <- matrix(1:20,
               nrow = 5,
               ncol = 4,
               byrow = TRUE,
               dimnames = list(
                 c("R1","R2","R3","R4","R5"),
                 c("C1","C2","C3","C4")
               ))

# 3x3 Matrix (Filled by Columns)
mat2 <- matrix(1:9,
               nrow = 3,
               ncol = 3,
               byrow = FALSE,
               dimnames = list(
                 c("R1","R2","R3"),
                 c("C1","C2","C3")
               ))

# 2x2 Matrix (Filled by Rows)
mat3 <- matrix(1:4,
               nrow = 2,
               ncol = 2,
               byrow = TRUE,
               dimnames = list(
                 c("R1","R2"),
                 c("C1","C2")
               ))

# Display Matrices
cat("5x4 Matrix (Filled by Rows):\n")
print(mat1)

cat("\n3x3 Matrix (Filled by Columns):\n")
print(mat2)

cat("\n2x2 Matrix (Filled by Rows):\n")
print(mat3)
