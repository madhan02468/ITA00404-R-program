1. Factors of a Number
n <- 12

for(i in 1:n) {
  if(n %% i == 0) {
    print(i)
  }
}
OUTPUT:
[1] 1
[1] 2
[1] 3
[1] 4
[1] 6
[1] 12

2. Generate 10 Random Integers Between -50 and 50
set.seed(10)

x <- sample(-50:50, 10)

print(x)
OUTPUT:
[1] -11  35 -30  42 -19   7  -2  28 -45  16

3. FizzBuzz from 1 to 100
for(i in 1:100) {
  if(i %% 15 == 0)
    print("FizzBuzz")
  else if(i %% 3 == 0)
    print("Fizz")
  else if(i %% 5 == 0)
    print("Buzz")
  else
    print(i)
}
OUTPUT :
[1] 1
[1] 2
[1] "Fizz"
[1] 4
[1] "Buzz"
[1] "Fizz"
...
[1] 14
[1] "FizzBuzz"
...
[1] 100
