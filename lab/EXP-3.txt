# Input rows, columns, and tables
rows <- as.integer(readline("Enter number of rows: "))
cols <- as.integer(readline("Enter number of columns: "))
tables <- as.integer(readline("Enter number of tables: "))

# Create a 3D array
arr <- array(1:(rows * cols * tables),
             dim = c(rows, cols, tables))

# Display the array
cat("3D Array:\n")
print(arr)

Enter number of rows: 2
Enter number of columns: 3
Enter number of tables: 2
